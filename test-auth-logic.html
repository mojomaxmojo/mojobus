// Mojo & Susanne Pubkeys from relays.ts const MOJO_PUBKEY = '4d584dab7c880a9809e7df0476d745bfe9a3fe91a1c062bc1fec024e0b5e1f1f'; const SUSANNE_PUBKEY = '94ebd1c0940881de438b7f3c532b73e0d4d6c6b0160d3fe0b8a55fe49d477bd4'; function isAuthorizedUser(pubkey: string): boolean { return pubkey === MOJO_PUBKEY || pubkey === SUSANNE_PUBKEY; } if (user && user.pubkey && !isAuthorizedUser(user.pubkey)) { return <div>Zugriff beschränkt</div>; }

✓ Zeigt "Zugriff beschränkt" für nicht-autorisierte User

Test 4: Authorized User (Mojo or Susanne)

if (user && user.pubkey && isAuthorizedUser(user.pubkey)) { return <HaushaltsbuchContent />; }

✓ Zeigt Haushaltsbuch für autorisierte User

Mögliche Probleme:

  • 1. useCurrentUser() gibt immer undefined zurück: Das Hook lädt noch
  • 2. Nostr Login nicht korrekt konfiguriert: NIP-07 Extension fehlt
  • 3. Pubkey-Mismatch: Eingeloggter User hat andere Pubkey als erwartet
  • 4. React State nicht aktualisiert: User-Objekt ändert sich nach Login

Debugging Steps:

  1. Browser Console öffnen (F12)
  2. Prüfen ob window.nostr existiert (NIP-07)
  3. useCurrentUser() Hook debuggen
  4. User-Pubkey mit Mojo/Susanne Pubkeys vergleichen
  5. LocalStorage prüfen auf nostr:login key
